38  enum CTRLTimingValues
39  {
40  // Protocol engine clock: 40 MHz CAN0_CLK from MC_CGM MUX_3 (see
41  // bspClock). The S32K344 FlexCAN has no CTRL1.CLKSRC bit, the PE
42  // clock is always the CANn_CLK.
43  // PRESDIV = 24 -> fTQ = fCLK / 25 = 1.6 MHz
44  // RJW = 0, PSEG1 = 4, PSEG2 = 2, PROPSEG = 6 -> Bittime = 16 TQ
45  CTRL_TIMING_LOWSPEED = (24U << 24) | (0U << 22) | (4U << 19) | (2U << 16) | (6U << 0),
46  // PRESDIV = 4 -> fTQ = fCLK / 5 = 8 MHz
47  // RJW = 0, PSEG1 = 4, PSEG2 = 2, PROPSEG = 6 -> Bittime = 16 TQ
48  CTRL_TIMING_HIGHSPEED = (4U << 24) | (0U << 22) | (4U << 19) | (2U << 16) | (6U << 0),
49  // PRESDIV = 1 -> fTQ = fCLK / 2 = 20 MHz
50  // RJW = 1, PSEG1 = 6, PSEG2 = 3, PROPSEG = 7 -> Bittime = 20 TQ
51  CTRL_TIMING_1MBAUD = (1U << 24) | (1U << 22) | (6U << 19) | (3U << 16) | (7U << 0)
52  };

291  struct MessageBuffer8Byte
292  {
293  union
294  { // FLAGS Register
295  vuint32_t R;
296 
297  struct
298  {
299  // Free running timestamp.
300  vuint32_t TIMESTAMP : 16;
301  // Length of data.
302  vuint32_t DLC : 4;
303  // Remote Transmission Request.
304  vuint32_t RTR : 1;
305  // ID Extended Bit. 1 extended, 0 standard.
306  vuint32_t IDE : 1;
307  // Substitute Remote Request
308  vuint32_t SRR : 1;
309  vuint32_t unnamed0 : 1;
310  vuint32_t CODE : 4;
311  vuint32_t unnamed1 : 1;
312  // Error State Indicator. Indicates if tx node is in error.
313  vuint32_t ESI : 1;
314  // Bit Rate Switch. Defines bit rate switch for CAN FD.
315  vuint32_t BRS : 1;
316  // Extended Data Length. Distinguishes between CAN and CAN FD.
317  vuint32_t EDL : 1;
318  } B;
319  } FLAGS;
320 
321  union
322  { // ID Register
323  vuint32_t R;
324 
325  struct
326  {
327  vuint32_t ID_EXT : 18;
328  vuint32_t ID_STD : 11;
329  vuint32_t PRIO : 3;
330  } B;
331  } ID;
332 
333  union
334  { // Frame payload
335  vuint8_t B[8];
336  vuint16_t H[4];
337  vuint32_t W[2];
338  } DATA;
339  };
340 
341  // The S32K344 device header does not expose the message buffer RAM as a
342  // struct member (the S32K148 header had a RAMn array); it starts at
343  // offset 0x80 and holds 96 message buffers of 16 bytes each.
344  static uint32_t const MESSAGE_BUFFER_RAM_OFFSET = 0x80U;
345  static uint32_t const MESSAGE_BUFFER_RAM_WORDS = (96U * 16U) / sizeof(uint32_t);
346 
347  MessageBuffer8Byte volatile& messageBuffer(uint8_t const bufIdx)
348  {
349  MessageBuffer8Byte volatile* msgBuffers = reinterpret_cast<MessageBuffer8Byte volatile*>(
350  fConfig.baseAddress + MESSAGE_BUFFER_RAM_OFFSET);
351  return msgBuffers[bufIdx];
352  }
